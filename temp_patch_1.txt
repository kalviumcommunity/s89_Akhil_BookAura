  // Reload language on path change
  useEffect(() => {
    const savedLang = localStorage.getItem('translate-lang') || 'en';
    setCurrentLang(savedLang);
    if (savedLang !== 'en') {
      setTimeout(() => translatePage(savedLang), 1000);
    }
  }, [location.pathname]);

  // Forcefully remove Google Translate banner and tooltip elements periodically
  useEffect(() => {
    const interval = setInterval(() => {
      const bannerFrames = document.querySelectorAll('.goog-te-banner-frame, .goog-te-balloon-frame, .goog-te-tooltip, .skiptranslate');
      bannerFrames.forEach(el => {
        if (el.style.display !== 'none') {
          el.style.display = 'none';
        }
        if (el.parentNode) {
          el.parentNode.removeChild(el);
        }
      });
    }, 500);

    return () => clearInterval(interval);
  }, []);
